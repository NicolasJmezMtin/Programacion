import java.util.Scanner;

public class SumaAcumuladaWhile {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int suma = 0;
        int num = 0;

        System.out.print("Introduce un número entero (negativo para terminar): ");
        num = sc.nextInt();

        while (num >= 0) {
            suma = suma + num;
            System.out.println("Suma acumulada: " + suma);

            System.out.print("Introduce un número entero (negativo para terminar): ");
            num = sc.nextInt();
        }

        System.out.println("Has introducido un número negativo. Total final: " + suma);

        sc.close();
    }
}
