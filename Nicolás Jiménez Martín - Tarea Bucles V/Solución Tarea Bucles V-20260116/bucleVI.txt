import java.util.Scanner;

public class InventarioLibreriaSinArrayList {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        final int MAX_LIBROS = 100; // Límite máximo de libros en el inventario
        String[] titulos = new String[MAX_LIBROS];
        String[] autores = new String[MAX_LIBROS];
        int[] aniosPublicacion = new int[MAX_LIBROS];
        double[] precios = new double[MAX_LIBROS];
        int contadorLibros = 0;
        int opcion;

        do {
            System.out.println("\nMenú de Librería:");
            System.out.println("1. Agregar nuevo libro");
            System.out.println("2. Mostrar todos los libros");
            System.out.println("3. Buscar libro por título");
            System.out.println("4. Salir");
            System.out.print("Seleccione una opción: ");
            opcion = scanner.nextInt();
            scanner.nextLine(); // Consumir el salto de línea

            if (opcion == 1) {
                if (contadorLibros < MAX_LIBROS) {
                    System.out.print("Ingrese el título del libro: ");
                    titulos[contadorLibros] = scanner.nextLine();
                    System.out.print("Ingrese el autor: ");
                    autores[contadorLibros] = scanner.nextLine();
                    System.out.print("Ingrese el año de publicación: ");
                    aniosPublicacion[contadorLibros] = scanner.nextInt();
                    System.out.print("Ingrese el precio: ");
                    precios[contadorLibros] = scanner.nextDouble();

                    if (aniosPublicacion[contadorLibros] > 0 && precios[contadorLibros] > 0) {
                        contadorLibros++;
                        System.out.println("Libro agregado correctamente.");
                    } else {
                        System.out.println("Error: Año de publicación y precio deben ser valores positivos.");
                    }
                } else {
                    System.out.println("Inventario lleno, no se pueden agregar más libros.");
                }
            } else if (opcion == 2) {
                System.out.println("\nLista de libros en el inventario:");
                if (contadorLibros == 0) {
                    System.out.println("No hay libros en el inventario.");
                } else {
                    for (int i = 0; i < contadorLibros; i++) {
                        System.out.println("Título: " + titulos[i]);
                        System.out.println("Autor: " + autores[i]);
                        System.out.println("Año de publicación: " + aniosPublicacion[i]);
                        System.out.println("Precio: $" + precios[i]);
                        System.out.println("-----------------------------");
                    }
                }
            } else if (opcion == 3) {
                System.out.print("Ingrese el título del libro a buscar: ");
                String tituloBuscar = scanner.nextLine();
                boolean encontrado = false;
                int i = 0;

                while (i < contadorLibros) {
                    if (titulos[i].equalsIgnoreCase(tituloBuscar)) {
                        System.out.println("\nLibro encontrado:");
                        System.out.println("Título: " + titulos[i]);
                        System.out.println("Autor: " + autores[i]);
                        System.out.println("Año de publicación: " + aniosPublicacion[i]);
                        System.out.println("Precio: $" + precios[i]);
                        encontrado = true;
                        break;
                    }
                    i++;
                }

                if (!encontrado) {
                    System.out.println("El libro no está en el inventario.");
                }
            } else if (opcion == 4) {
                System.out.println("Saliendo del programa...");
            } else {
                System.out.println("Opción no válida. Intente nuevamente.");
            }

        } while (opcion != 4);

        scanner.close();
    }
}
